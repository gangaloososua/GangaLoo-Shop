3 O pega este código en la consola del navegador

En la página Shein, pulsa F12 → pestaña Console → pega esto y pulsa Enter:

(function() { const results = []; const seen = new Set(); // Strategy 1: find all tags that wrap an - broadest possible approach document.querySelectorAll('a').forEach(a => { const href = a.href || ''; if (!href || href === location.href) return; const img = a.querySelector('img') || a.closest('[class]')?.querySelector('img'); if (!img) return; const imgSrc = img.src || img.dataset.src || img.dataset.lazySrc || img.dataset.original || img.getAttribute('data-src') || ''; if (!imgSrc || imgSrc.startsWith('data:') || imgSrc.includes('placeholder')) return; // Get name - walk up to find a container with text let name = '', price = ''; let container = a.parentElement; for (let i = 0; i < 5; i++) { if (!container) break; const txt = container.innerText || ''; if (txt.length > 5 && txt.length < 300) { name = txt.split('\n')[0].trim(); break; } container = container.parentElement; } const key = imgSrc.split('?')[0]; if (seen.has(key)) return; seen.add(key); results.push({ image_url: imgSrc.split('?')[0].replace(/(_\d+x\d+)/, '_405x552'), affiliate_url: href, name: name.substring(0, 120) || 'Producto Shein', price: '' }); }); // Strategy 2: all imgs with shein CDN urls if (results.length === 0) { document.querySelectorAll('img').forEach(img => { const src = img.src || img.dataset.src || img.dataset.lazySrc || ''; if (!src || src.startsWith('data:')) return; if (!src.includes('ltwebstatic') && !src.includes('shein') && !src.includes('img.')) return; const key = src.split('?')[0]; if (seen.has(key)) return; seen.add(key); // Find closest link let link = img.closest('a'); if (!link) { let el = img.parentElement; for (let i = 0; i < 6; i++) { if (!el) break; const a = el.querySelector('a[href]'); if (a) { link = a; break; } el = el.parentElement; } } results.push({ image_url: src.split('?')[0].replace(/(_\d+x\d+)/, '_405x552'), affiliate_url: link?.href || '', name: img.alt || 'Producto Shein', price: '' }); }); } // Strategy 3: dump all page text + links as debug if (results.length === 0) { const allLinks = [...document.querySelectorAll('a[href]')].map(a => a.href).filter(h => h.includes('shein') || h.includes('onelink')); const allImgs = [...document.querySelectorAll('img')].map(i => i.src).filter(s => s && !s.startsWith('data:')); const debug = { url: location.href, title: document.title, bodyLength: document.body.innerHTML.length, linkCount: allLinks.length, imgCount: allImgs.length, sampleLinks: allLinks.slice(0, 10), sampleImgs: allImgs.slice(0, 10), bodySnippet: document.body.innerText.substring(0, 500) }; const blob = new Blob([JSON.stringify(debug, null, 2)], {type: 'application/json'}); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'shein-debug.json'; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); alert('❌ No se encontraron productos.\n\nSe descargó shein-debug.json — envíalo a GangaLoo para analizar la estructura de la página.\n\nLinks encontrados: ' + allLinks.length + '\nImágenes encontradas: ' + allImgs.length); return; } // Download JSON const blob = new Blob([JSON.stringify(results, null, 2)], {type: 'application/json'}); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'shein-products.json'; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); // Show overlay const ovl = document.createElement('div'); ovl.style.cssText = 'position:fixed;inset:0;background:rgba(0,0,0,.8);z-index:999999;display:flex;align-items:center;justify-content:center;padding:20px;'; ovl.innerHTML = `

✅ ${results.length} productos encontrados

shein-products.json descargado. Súbelo al importador.

${results.slice(0,5).map(p => `
${p.name}
`).join('')} ${results.length > 5 ? `
... y ${results.length-5} más
` : ''}
`; document.body.appendChild(ovl); })();
