3 O pega este código en la consola del navegador

En la página Shein, pulsa F12 → pestaña Console → pega esto y pulsa Enter:

(function() { const results = []; const seen = new Set(); const pageUrl = location.href; // Extract pageId from URL to build affiliate links const pageIdMatch = pageUrl.match(/pageId=([^&]+)/); const pageId = pageIdMatch ? pageIdMatch[1] : ''; const identifierMatch = pageUrl.match(/identifier=([^&]+)/); const identifier = identifierMatch ? identifierMatch[1] : ''; // Find all images with Shein CDN URLs document.querySelectorAll('img').forEach(img => { const src = img.src || img.dataset.src || img.dataset.lazySrc || img.getAttribute('data-src') || img.getAttribute('data-original') || ''; if (!src || src.startsWith('data:') || src.length < 20) return; if (!src.includes('ltwebstatic') && !src.includes('shein') && !src.includes('kwcdn') && !src.includes('img.')) return; const key = src.split('?')[0]; if (seen.has(key)) return; seen.add(key); // Walk up DOM to find: link, goods-id, product id, name let link = '', name = '', goodsId = '', discount = ''; let el = img; for (let i = 0; i < 10; i++) { if (!el) break; // Check for data attributes with product ID const gid = el.dataset.goodsId || el.dataset.productId || el.dataset.id || el.getAttribute('data-goods-id') || el.getAttribute('data-spu-id') || ''; if (gid && !goodsId) goodsId = gid; // Check for anchor if (el.tagName === 'A' && el.href && !el.href.includes('javascript')) { link = el.href; } // Check parent for anchor if (!link) { const a = el.querySelector && el.querySelector('a[href]'); if (a && a.href && !a.href.includes('javascript')) link = a.href; } el = el.parentElement; } // Get name from alt or nearby text name = img.alt || img.title || ''; if (!name) { let container = img.parentElement; for (let i = 0; i < 6; i++) { if (!container) break; const txt = (container.innerText || '').trim(); const lines = txt.split('\n').map(l => l.trim()).filter(l => l.length > 5 && l.length < 150); if (lines.length) { name = lines[0]; break; } container = container.parentElement; } } // Build affiliate link - use captured link or construct from goods ID or use page URL let affiliateUrl = link; if (!affiliateUrl || affiliateUrl === pageUrl) { if (goodsId) { // Build Shein product link with affiliate params affiliateUrl = `https://www.shein.com/product/p-${goodsId}.html?ref=affiliate`; } else if (link) { affiliateUrl = link; } else { affiliateUrl = pageUrl; // fallback to affiliate page } } // Make sure link is absolute and valid if (affiliateUrl.startsWith('/')) affiliateUrl = 'https://www.shein.com' + affiliateUrl; // Extract discount/price from nearby DOM const container2 = img.closest('[class]'); if (container2) { const priceEl = container2.querySelector('[class*="price"], [class*="discount"], [class*="off"]'); if (priceEl) discount = priceEl.innerText?.trim() || ''; } results.push({ image_url: key.replace(/(_\d+x\d+)/, '_405x552'), affiliate_url: affiliateUrl, goods_id: goodsId, name: (name || 'Producto Shein').substring(0, 120), price: discount }); }); if (!results.length) { // Debug dump const dbg = { url: pageUrl, title: document.title, bodyLength: document.body.innerHTML.length, allLinks: [...document.querySelectorAll('a[href]')].map(a=>({href:a.href,text:a.innerText?.substring(0,40)})).slice(0,20), allImgs: [...document.querySelectorAll('img')].map(i=>({src:i.src,alt:i.alt,ds:i.dataset})).slice(0,20), dataAttrs: [...document.querySelectorAll('[data-goods-id],[data-product-id],[data-spu-id]')].map(e=>({tag:e.tagName,data:JSON.stringify(e.dataset)})).slice(0,20), bodyText: document.body.innerText.substring(0, 800) }; const b = new Blob([JSON.stringify(dbg,null,2)],{type:'application/json'}); const u = URL.createObjectURL(b); const a = document.createElement('a'); a.href=u; a.download='shein-debug.json'; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(u); alert('❌ 0 productos.\nSe descargó shein-debug.json — compártelo para analizar la estructura.'); return; } // Download JSON const blob = new Blob([JSON.stringify(results,null,2)],{type:'application/json'}); const url2 = URL.createObjectURL(blob); const a2 = document.createElement('a'); a2.href=url2; a2.download='shein-products.json'; document.body.appendChild(a2); a2.click(); document.body.removeChild(a2); URL.revokeObjectURL(url2); // Overlay const ovl = document.createElement('div'); ovl.style.cssText='position:fixed;inset:0;background:rgba(0,0,0,.85);z-index:999999;display:flex;align-items:center;justify-content:center;padding:20px;'; ovl.innerHTML=`

shein-products.json descargado. Revisa los links abajo:

'}

${p.affiliate_url.substring(0,60)}
`).join('')} ${results.length>5?`
... y ${results.length-5} más en el JSON
`:''}
`; document.body.appendChild(ovl); })();
